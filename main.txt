local function mountVeilsGui(root)
    local candidates={}
    if type(gethui)=='function' then
        local ok,parent=pcall(gethui)
        if ok and typeof(parent)=='Instance' then table.insert(candidates,parent) end
    end
    local ok,core=pcall(function() return game:GetService('CoreGui') end)
    if ok then table.insert(candidates,core) end
    local player=game:GetService('Players').LocalPlayer
    if player then
        local playerGui=player:FindFirstChildOfClass('PlayerGui') or player:WaitForChild('PlayerGui',10)
        if playerGui then table.insert(candidates,playerGui) end
    end
    for _,parent in ipairs(candidates) do
        local mounted=pcall(function() root.Parent=parent end)
        if mounted and root.Parent==parent then return end
    end
    error('No usable UI parent was found; run after joining an experience')
end
local function showStartupStatus(message,failed)
    if not startupOverlay then
        startupOverlay=Instance.new('ScreenGui')
        startupOverlay.Name='VeilsStartupStatus'
        startupOverlay.ResetOnSpawn=false
        startupOverlay.IgnoreGuiInset=true
        startupOverlay.DisplayOrder=10000
        mountVeilsGui(startupOverlay)
        for _,child in ipairs(startupOverlay.Parent:GetChildren()) do
            if child~=startupOverlay and child.Name=='VeilsStartupStatus' and child:IsA('ScreenGui') then child:Destroy() end
        end
        local label=Instance.new('TextLabel')
        label.Name='Status'
        label.Position=UDim2.fromOffset(16,50)
        label.Size=UDim2.new(0,math.min(560,math.max(260,(workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize.X or 600)-32)),0,110)
        label.BackgroundColor3=Color3.fromRGB(22,22,25)
        label.TextColor3=Color3.fromRGB(240,240,244)
        label.TextSize=14
        label.Font=Enum.Font.Gotham
        label.TextWrapped=true
        label.ZIndex=100
        label.Parent=startupOverlay
        local corner=Instance.new('UICorner'); corner.CornerRadius=UDim.new(0,12); corner.Parent=label
    end
    startupOverlay.Status.Text=message
    startupOverlay.Status.TextColor3=failed and Color3.fromRGB(255,130,140) or Color3.fromRGB(240,240,244)
end

do
    local layer=make('Frame',{Name='AccentSnow',Size=UDim2.fromScale(1,1),BackgroundTransparency=1,
        ClipsDescendants=true,Active=false,ZIndex=40},window)
    local random=Random.new()

    local elapsed,lastColor=0,nil
    for i=1,32 do
        local diameter=random:NextNumber(2,4)
        local dot=make('Frame',{Name='Flake',Size=UDim2.fromOffset(diameter,diameter),
            BackgroundColor3=color(S.Accent),BackgroundTransparency=random:NextNumber(.48,.78),
            BorderSizePixel=0,Active=false,ZIndex=41},layer); round(dot,99)
        flakes[i]={dot=dot,x=random:NextNumber(0,980),y=random:NextNumber(-12,620),
            speed=random:NextNumber(17,38),sway=random:NextNumber(3,11),phase=random:NextNumber(0,math.pi*2)}
    end
    updateMenuSnow=function(dt)
        layer.Visible=window.Visible
        if not window.Visible then return end
        dt=math.clamp(dt,0,.05); elapsed=elapsed+dt
        if lastColor~=S.Accent then
            lastColor=S.Accent
            for _,flake in ipairs(flakes) do flake.dot.BackgroundColor3=color(S.Accent) end
        end
        for _,flake in ipairs(flakes) do
            flake.y=flake.y+flake.speed*dt
            if flake.y>628 then flake.y=-8; flake.x=random:NextNumber(0,980) end
            local x=flake.x+math.sin(elapsed*.7+flake.phase)*flake.sway
            flake.dot.Position=UDim2.fromOffset(x,flake.y)
        end
    end
end

do
    -- Keep the profile keys so existing V1/V2 settings continue to load.
    local weaponNames={revolver=true,gun=true,pistol=true,glock=true,silencer=true,silencedpistol=true,
        shotgun=true,doublebarrel=true,doublebarrelsg=true,db=true,tacticalshotgun=true,drumshotgun=true,
        silencerar=true,smg=true,silencersmg=true,p90=true,aug=true,ak47=true,ar=true,ar15=true,
        rifle=true,assaultrifle=true,lmg=true,drumgun=true,uzi=true,mp5=true,
        sniper=true,sniperrifle=true,rpg=true,rocketlauncher=true,flamethrower=true}
    local function isWeapon(tool)
        local name=(tool.Name:match('%[([^%]]+)%]') or tool.Name):lower():gsub('[^%w]','')
        if weaponNames[name] then return true end
        -- Custom guns may expose ammo rather than a standard weapon name.
        for _,key in ipairs({'Ammo','Ammunition','Clip','Magazine'}) do
            local value=tool:FindFirstChild(key,true)
            if value and (value:IsA('IntValue') or value:IsA('NumberValue')) then return true end
            if type(tool:GetAttribute(key))=='number' then return true end
        end
        return false
    end
    local pressed=false
    local lastShot=0
    local activeTool
    local triggerDown=false
    local lastMode
    local function release()
        if activeTool and activeTool.Parent then pcall(function() activeTool:Deactivate() end) end
        activeTool=nil; triggerDown=false
    end
    connect(Input.InputBegan,function(input,processed)
        if input.UserInputType~=Enum.UserInputType.MouseButton1 or processed or Input:GetFocusedTextBox() then return end
        local mouse=Input:GetMouseLocation()
        local pos,size=window.AbsolutePosition,window.AbsoluteSize
        if window.Visible and mouse.X>=pos.X and mouse.Y>=pos.Y and mouse.X<=pos.X+size.X and mouse.Y<=pos.Y+size.Y then return end
        pressed=true; lastShot=0
    end)
    connect(Input.InputEnded,function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 then pressed=false; release() end
    end)
    connect(Input.WindowFocusReleased,function() pressed=false; release() end)
    connect(Run.Heartbeat,function()
        if not alive or not (S.RevolverAutoFire or S.RevolverAutoFireV2) or not pressed or not clientFocused or Input:GetFocusedTextBox() then release(); return end
        local character=LP.Character
        local humanoid=character and character:FindFirstChildOfClass('Humanoid')
        local tool=character and character:FindFirstChildOfClass('Tool')
        if not humanoid or humanoid.Health<=0 or not tool or not isWeapon(tool) then release(); return end
        local mode=S.RevolverAutoFireV2 and 'V2' or 'V1'
        if lastMode~=mode or (activeTool and activeTool~=tool) then release(); lastMode=mode end
        -- V2 gives the trigger a complete released frame between presses.
        if mode=='V2' and triggerDown then
            release()
            return
        end
        local now=os.clock()
        local interval=math.clamp(tonumber(S.RevolverFireInterval) or 0,0,.5)
        if tool.Enabled and now-lastShot>=interval then
            lastShot=now; activeTool=tool
            -- Use normal tool input; the weapon retains its own cooldown and ammo checks.
            if mode=='V2' then
                tool:Activate(); triggerDown=true
            else
                tool:Deactivate(); tool:Activate()
            end
        end
    end)
    connect(gui.Destroying,release)
end

local function normalizeAnimationID(value)
    if type(value)~='string' then return nil end
    local id=value:match('^%s*(%d+)%s*$') or value:match('^rbxassetid://(%d+)$')
        or value:match('/catalog/(%d+)') or value:match('/store/asset/(%d+)')
    if id and #id<=19 and id:find('[1-9]') then return id end
    return nil
end

local function applyProfile(data)
    for k,v in pairs(data) do
        if k~='Profile' and S[k]~=nil and type(v)==type(S[k]) then
            if type(v)=='boolean' then S[k]=v
            elseif type(v)=='number' and v==v and math.abs(v)<1e6 then S[k]=v

-- Visual-only rigs: anchor every part, including late-loading accessory handles.
local function freezeVisualPart(x)
    if x:IsA('BasePart') then
        x.Anchored=true; x.CanCollide=false; x.CanTouch=false; x.CanQuery=false
        x.Massless=true; x.AssemblyLinearVelocity=Vector3.zero; x.AssemblyAngularVelocity=Vector3.zero
    elseif x:IsA('LuaSourceContainer') or x:IsA('Animator') or x:IsA('BodyMover') or x:IsA('Constraint') then
        x:Destroy()
    end
end
local function solveVisualRig(model)
    local root=model:FindFirstChild('HumanoidRootPart'); if not root then return end
    local visited={[root]=true}
    local joints={}
    for _,x in ipairs(model:GetDescendants()) do
        if x:IsA('Motor6D') and x.Part0 and x.Part1 then table.insert(joints,x) end
    end
    -- Resolve the joint tree explicitly: anchored rigs do not receive solver-driven joint positioning.
    for _=1,#joints+1 do
        local changed=false
        for _,joint in ipairs(joints) do
            if visited[joint.Part0] and not visited[joint.Part1] then
                joint.Part1.CFrame=joint.Part0.CFrame*joint.C0*joint.Transform*joint.C1:Inverse()
                visited[joint.Part1]=true; changed=true
            end
        end
        if not changed then break end
    end
end
local function syncVisualAccessories(model)
    local attachments={}
    for _,body in ipairs(model:GetChildren()) do
        if body:IsA('BasePart') then
            for _,x in ipairs(body:GetChildren()) do if x:IsA('Attachment') then attachments[x.Name]=x end end
        end
    end
    for _,accessory in ipairs(model:GetChildren()) do
        if accessory:IsA('Accessory') then
            local handle=accessory:FindFirstChild('Handle')
            if handle and handle:IsA('BasePart') then
                local placed=false
                for _,attachment in ipairs(handle:GetChildren()) do
                    if attachment:IsA('Attachment') and attachments[attachment.Name] then
                        handle.CFrame=attachments[attachment.Name].WorldCFrame*attachment.CFrame:Inverse(); placed=true; break
                    end
                end
                if not placed then
                    local weld=handle:FindFirstChild('AccessoryWeld')
                    if weld and weld:IsA('Weld') and weld.Part0 and weld.Part1 then
                        if weld.Part0==handle then handle.CFrame=weld.Part1.CFrame*weld.C1*weld.C0:Inverse()
                        elseif weld.Part1==handle then handle.CFrame=weld.Part0.CFrame*weld.C0*weld.C1:Inverse() end
                    end
                end
            end
        end
    end
end
local function prepareVisualModel(model)
    for _,x in ipairs(model:GetDescendants()) do freezeVisualPart(x) end
    connect(model.DescendantAdded,freezeVisualPart)
    local h=model:FindFirstChildOfClass('Humanoid')
    if h then
        h.DisplayDistanceType=Enum.HumanoidDisplayDistanceType.None
        h.PlatformStand=true; h.AutoRotate=false
        pcall(function() h.EvaluateStateMachine=false end)
    end
    solveVisualRig(model); syncVisualAccessories(model)
end
local function visualBodyBounds(model)
    local low=Vector3.new(math.huge,math.huge,math.huge); local high=-low
    for _,part in ipairs(model:GetChildren()) do
        if part:IsA('BasePart') and part.Name~='HumanoidRootPart' then
            for x=-1,1,2 do for y=-1,1,2 do for z=-1,1,2 do
                local p=part.CFrame:PointToWorldSpace(part.Size*Vector3.new(x,y,z)/2)
                low=Vector3.new(math.min(low.X,p.X),math.min(low.Y,p.Y),math.min(low.Z,p.Z))
                high=Vector3.new(math.max(high.X,p.X),math.max(high.Y,p.Y),math.max(high.Z,p.Z))
            end end end
        end
    end
    if low.X==math.huge then return CFrame.new(),Vector3.new(4,6,2) end
    return CFrame.new((low+high)/2),high-low
end

local function beginMenuDrag(i)
    local kind=i.UserInputType
    if kind~=Enum.UserInputType.MouseButton1 and kind~=Enum.UserInputType.Touch then return end

    local focused=Input:GetFocusedTextBox()
    if focused then focused:ReleaseFocus() end
    dragState={input=i,touch=kind==Enum.UserInputType.Touch,origin=p,
        position=window.Position,absolute=window.AbsolutePosition}
end
local function moveMenuDrag(i)
    local drag=dragState; if not drag then return end
    if not window.Visible then dragState=nil; return end
    if drag.touch then
        if i~=drag.input then return end
    elseif i.UserInputType~=Enum.UserInputType.MouseMovement then return
    elseif not Input:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then dragState=nil; return end
    local delta=Vector2.new(i.Position.X,i.Position.Y)-drag.origin
    local topLeft=drag.absolute+delta
    local camera=workspace.CurrentCamera
    if camera and camera.ViewportSize.X>0 and camera.ViewportSize.Y>0 then
        local view=camera.ViewportSize
        topLeft=Vector2.new(math.clamp(topLeft.X,80-window.AbsoluteSize.X,math.max(0,view.X-80)),
            math.clamp(topLeft.Y,0,math.max(0,view.Y-40)))
    end
    delta=topLeft-drag.absolute
    window.Position=UDim2.new(drag.position.X.Scale,drag.position.X.Offset+delta.X,
        drag.position.Y.Scale,drag.position.Y.Offset+delta.Y)
end
local function endMenuDrag(i)

end

-- Resolve both Marketplace emote wrappers once per flight session.
local flightAnimationSignature
local flightAnimations={}
local flightAnimationGeneration=0
local flightAnimationMode
stopFlightAnimation=function()
    flightAnimationGeneration=flightAnimationGeneration+1
    for _,entry in pairs(flightAnimations) do
        if entry.track then pcall(function() entry.track:Stop(.15); entry.track:Destroy() end) end
        if entry.asset then entry.asset:Destroy() end
    end
    flightAnimations={}; flightAnimationMode=nil; flightAnimationSignature=nil
end
local function ensureFlightAnimation(h,mode)
    if flightAnimations[mode] then return end
    local entry={loading=true,restartAt=0,startedAt=0}
    flightAnimations[mode]=entry
    local emoteID=mode=='Moving' and S.FlyAnimationID or S.HoverAnimationID
    local generation=flightAnimationGeneration
    local function current()
        return alive and generation==flightAnimationGeneration and S.Fly
            and (not S.Desync or (desync and desync.proxyHumanoid==h))
            and flightHumanoid==h and h.Parent~=nil and h.Health>0
            and emoteID==(mode=='Moving' and S.FlyAnimationID or S.HoverAnimationID)
    end
    task.spawn(function()
        local objects,asset,track
        local ok,err=pcall(function()
            local animator=h:FindFirstChildOfClass('Animator')
            assert(animator,'Character Animator missing')
            objects=game:GetObjects('rbxassetid://'..emoteID)
            assert(type(objects)=='table','Object loading did not return an asset')
            local underlying
            local function inspect(x)
                if x:IsA('KeyframeSequence') or x:IsA('CurveAnimation') then
                    underlying='rbxassetid://'..emoteID
                elseif x:IsA('Animation') then
                    local uri=x.AnimationId
                    local number=uri:match('rbxassetid://(%d+)') or uri:match('[?&]id=(%d+)') or uri:match('^(%d+)$')
                    if number and number~=emoteID then underlying='rbxassetid://'..number end
                end
            end
            for _,object in ipairs(objects) do
                inspect(object)
                for _,child in ipairs(object:GetDescendants()) do inspect(child) end
            end
            assert(underlying,'The emote wrapper did not expose an embedded AnimationId')
            if not current() then return end
            asset=Instance.new('Animation'); asset.AnimationId=underlying
            track=animator:LoadAnimation(asset)
            track.Priority=Enum.AnimationPriority.Action4; track.Looped=true
            print('[Veils Universe] Resolved '..mode..' animation: '..underlying)
        end)
        if objects then for _,object in ipairs(objects) do pcall(function() object:Destroy() end) end end
        if not ok or not current() then
            if track then pcall(function() track:Stop(.1); track:Destroy() end) end
            if asset then asset:Destroy() end
            if current() then
                entry.loading=false; entry.failed=true
                notify(mode..' animation failed to resolve; flight controls remain active')
                warn('[Veils Universe '..mode..' animation] '..tostring(err))
            end
            return
        end
        entry.track=track; entry.asset=asset; entry.loading=false
        -- Playback is selected by movement state, never by asynchronous load completion.
    end)
end
local function updateFlightAnimation(h,moving)
    local signature=S.FlyAnimationID..':'..S.HoverAnimationID
    if flightAnimationSignature~=signature then stopFlightAnimation(); flightAnimationSignature=signature end
    ensureFlightAnimation(h,'Moving'); ensureFlightAnimation(h,'Hover')
    local desired=moving and 'Moving' or 'Hover'
    local entry=flightAnimations[desired]
    if not entry or not entry.track or entry.failed then return end
    local now=os.clock()
    if flightAnimationMode~=desired then
        local previous=flightAnimationMode and flightAnimations[flightAnimationMode]
        if previous and previous.track then previous.track:Stop(.2) end
        flightAnimationMode=desired
        entry.track:Play(.2,1,1); entry.startedAt=now; entry.restartAt=now+1
    elseif entry.track.Length==0 and now-entry.startedAt>8 then
        entry.failed=true; entry.track:Stop(.15); flightAnimationMode=nil
        notify(desired..' animation did not load in this experience')
    elseif entry.track.Length>0 and not entry.track.IsPlaying and now>entry.restartAt then
        entry.restartAt=now+1; entry.track:Play(.2,1,1)
    end
end
local function stopFlight()
    stopFlightAnimation()
    if flight and flight.Parent and flight.Parent:IsA('BasePart') then
        flight.Parent.AssemblyLinearVelocity=Vector3.zero
        flight.Parent.AssemblyAngularVelocity=Vector3.zero
    end
    if flight then flight:Destroy(); flight=nil end
    if flightOrientation then flightOrientation:Destroy(); flightOrientation=nil end
    if flightAttachment then flightAttachment:Destroy(); flightAttachment=nil end
    if flightHumanoid and flightHumanoid.Parent then
        flightHumanoid.AutoRotate=oldAutoRotate; flightHumanoid.PlatformStand=oldPlatform
    end
    flightHumanoid=nil
end
local function stopFollowing(message,restoreFly)
    local state=followState; followState=nil
    if not state then return end
    if restoreFly~=false then S.Fly=state.previousFly end
    stopFlight()
    if alive then refresh(); if message then notify(message) end end
end
local function findFollowTarget(query,exclude)
    local needle=query:lower():gsub('^@',''):gsub('^%(',''):gsub('%)$','')
    if needle=='random' then
        local candidates={}
        for _,player in ipairs(Players:GetPlayers()) do
            if player~=LP and player~=exclude and player.Parent==Players and rig(player) then table.insert(candidates,player) end
        end
        if #candidates==0 then return nil,'No other living players available' end
        return candidates[Random.new():NextInteger(1,#candidates)]
    end
    local exact,prefix={},{}
    for _,player in ipairs(Players:GetPlayers()) do
        if player~=LP then
            local name,display=player.Name:lower(),player.DisplayName:lower()
            if name==needle then return player end
            if display==needle then table.insert(exact,player) end
            if name:sub(1,#needle)==needle or display:sub(1,#needle)==needle then table.insert(prefix,player) end
        end
    end
    if #exact==1 then return exact[1] end
    if #exact>1 or #prefix>1 then return nil,'More than one match; use the full username' end
    if #prefix==1 then return prefix[1] end
    return nil,'Player not found'
end
local function startFollowing(query,mode)
    local target,err=findFollowTarget(query)
    if not target then notify(err); return end
    local character=rig(LP); local other=rig(target)
    if not character or not other then notify('Both characters must be alive for stand mode'); return end
    local previous=followState and followState.previousFly
    if not followState then previous=S.Fly end
    followState={target=target,previousFly=previous,localCharacter=character,targetCharacter=other,
        mode=mode or 'stand',phaseStart=os.clock(),lastMotionAt=-math.huge}
    S.Fly=true; refresh(); notify((mode=='orbit' and 'Orbiting @' or 'Stand linked to @')..target.Name..' - .unstand to release')
end
local function replaceStandTarget(state,exclude)
    if state.controllerUserId then return false end
    if not S.StandAutoReplace then return false end
    local target=findFollowTarget('random',exclude or state.target)
    if not target then return false end
    state.target=target; state.targetCharacter=target.Character
    state.phaseStart=os.clock(); state.lastMotionAt=-math.huge
    notify('Stand switched to @'..target.Name)
    return true
end
local function followGoal(state)
    if LP.Character~=state.localCharacter then return nil,'Stand ended: your character changed' end
    local character,humanoid,root=rig(state.target)
    if state.target.Parent~=Players or not character or character~=state.targetCharacter then
        if not replaceStandTarget(state,state.target) then return nil,'Stand ended: no replacement target available' end
        character,humanoid,root=rig(state.target)
    end
    if not character or not root then return nil,'Stand ended: target unavailable' end
    local goal
    if state.mode=='underground' then
        -- Begin below the normal stand slot, then rise into formation once.
        local startY=math.max(root.Position.Y-7,workspace.FallenPartsDestroyHeight+20)
        local endY=root.Position.Y+1.5
        if startY>=endY then return nil,'Stand entrance is too close to the void boundary' end
        state.entranceStartedAt=state.entranceStartedAt or os.clock()
        local progress=math.clamp((os.clock()-state.entranceStartedAt)/1.35,0,1)
        local eased=progress*progress*(3-2*progress)
        local slot=root.CFrame*CFrame.new(3,0,4.5)
        goal=slot+Vector3.new(0,startY+(endY-startY)*eased-slot.Position.Y,0)
        if progress<1 then return goal,Vector3.zero,false,true end
        state.mode='stand'; state.entranceStartedAt=nil
    elseif state.mode=='orbit' then
        local angle=(os.clock()-state.phaseStart)*.55+math.atan2(4.5,3)
        local radius=math.sqrt(3*3+4.5*4.5)
        goal=root.CFrame*CFrame.new(math.cos(angle)*radius,1.5,math.sin(angle)*radius)
    else goal=root.CFrame*CFrame.new(3,1.5,4.5) end
    local speed=Vector3.new(root.AssemblyLinearVelocity.X,0,root.AssemblyLinearVelocity.Z).Magnitude
    if speed>1.5 or humanoid.MoveDirection.Magnitude>.1 then state.lastMotionAt=os.clock() end
    local moving=os.clock()-state.lastMotionAt<.25
    return goal,root.AssemblyLinearVelocity,moving
end
local function changeStandWhitelist(argument,remove)
    local query=argument:match('^%s*(.-)%s*$')
    if query=='' or query:lower()=='list' then
        local names={}; for _,entry in pairs(standWhitelist) do table.insert(names,'@'..entry.Name) end
        table.sort(names); notify(#names>0 and ('Whitelisted: '..table.concat(names,', ')) or 'Stand whitelist is empty'); return
    end
    if remove and query:lower()=='all' then
        standWhitelist={}
        if followState and followState.controllerUserId then stopFollowing('Whitelist cleared; stand released') end
        notify('Stand whitelist cleared'); return
    end
    if query:lower()=='random' then notify('Use a specific username for the whitelist'); return end
    local player,err=findFollowTarget(query)
    local userId=player and player.UserId
    if remove and not userId then
        local needle=query:lower():gsub('^@','')
        for id,entry in pairs(standWhitelist) do if entry.Name:lower()==needle then userId=id; break end end
    end
    if not userId then notify(err or 'Player not found'); return end
    if remove then
        standWhitelist[userId]=nil
        if followState and followState.controllerUserId==userId then stopFollowing('Access revoked; stand released') end
        notify('Removed stand access')
    else
        standWhitelist[userId]={Name=player.Name}
        notify('@'..player.Name..' can use !stand / .stand and !unstand / .unstand')
    end
end
local function handleTrustedStandMessage(player,message)
    if not alive or not player or player==LP or player.Parent~=Players or not standWhitelist[player.UserId] then return end
    if type(message)~='string' then return end
    local command=message:lower():match('^%s*[!.](%a+)%s*$')
    if command=='stand' then
        local previous=followState
        startFollowing(player.Name,'underground')
        if followState and followState~=previous and followState.target==player then
            followState.controllerUserId=player.UserId
            notify('Stand entrance linked to @'..player.Name)
        end
    elseif command=='unstand' and followState and followState.controllerUserId==player.UserId then
        stopFollowing('@'..player.Name..' released stand')
    end
end
local function handleFollowCommand(message)
    if type(message)~='string' then return end
    local verb,argument=message:match('^%s*([./]%a+)%s*(.-)%s*$')
    if not verb then return end
    verb=verb:lower()
    if verb=='.standtest' or verb=='/standtest' then
        if argument=='' then notify('Usage: .standtest whitelisted_username'); return end
        local player,err=findFollowTarget(argument)
        if not player then notify(err); return end
        if not standWhitelist[player.UserId] then notify('Whitelist this player first'); return end
        handleTrustedStandMessage(player,'!stand')
        return
    end
    if verb=='.whitelist' or verb=='/whitelist' then changeStandWhitelist(argument,false); return end
    if verb=='.unwhitelist' or verb=='/unwhitelist' then changeStandWhitelist(argument,true); return end
    if verb=='.comend' or verb=='/comend' then
        if openCommandPalette then openCommandPalette() end
        return
    end
    if verb=='.stand' or verb=='/stand' or verb=='.orbit' or verb=='/orbit' then
        local mode=verb:sub(2)
        if argument=='' and mode=='orbit' and followState then argument=followState.target.Name end
        if argument=='' then notify('Usage: '..verb..' username (or random)'); return end
        startFollowing(argument,mode)
    elseif verb=='.unstand' or verb=='/unstand' or verb=='.unorbit' or verb=='/unorbit' then
        if followState then stopFollowing('Stand disabled') else notify('Stand is not active') end
    end
end

local function prepareStandCharacter(root)
    local state=followState
    if not state then return true end
    local character=rig(LP)
    if not root or not character then
        if not S.StandAutoResume then stopFollowing('Stand ended: your character unavailable'); return false end
        if not state.waitingRespawn then
            state.waitingRespawn=true
            stopFlight(); notify('Stand paused - waiting for your respawn')
        end
        return false
    end
    if state.waitingRespawn or character~=state.localCharacter then
        if not S.StandAutoResume then stopFollowing('Stand ended: your character changed'); return false end
        state.localCharacter=character; state.waitingRespawn=false
        state.phaseStart=os.clock(); state.lastMotionAt=-math.huge
        notify('Stand resumed - '..state.mode..' @'..state.target.Name)
    end
    return true
end
local function updateFlight(h,r)
    local cam=workspace.CurrentCamera
    if followState and not S.Fly then stopFollowing(nil,false) end
    if followState and not prepareStandCharacter(r) then stopFlight(); return end
    if not S.Fly or not r or not cam then stopFlight(); return end
    if not flight or flight.Parent~=r then
        stopFlight(); flightHumanoid=h; oldAutoRotate=h.AutoRotate; oldPlatform=h.PlatformStand
        h.AutoRotate=false; h.PlatformStand=true
        r.AssemblyLinearVelocity=Vector3.zero; r.AssemblyAngularVelocity=Vector3.zero
        flightAttachment=make('Attachment',{Name='VeilsFlightAttachment'},r)
        flight=make('LinearVelocity',{Attachment0=flightAttachment,RelativeTo=Enum.ActuatorRelativeTo.World,
            ForceLimitsEnabled=false,VelocityConstraintMode=Enum.VelocityConstraintMode.Vector,
            MaxForce=math.huge,VectorVelocity=Vector3.zero},r)
        flightOrientation=make('AlignOrientation',{Name='VeilsFlightOrientation',Attachment0=flightAttachment,
            Mode=Enum.OrientationAlignmentMode.OneAttachment,RigidityEnabled=true,
            MaxTorque=math.huge,Responsiveness=200,CFrame=r.CFrame.Rotation},r)
    end
    -- The movement and orientation constraints own the assembly throughout flight.
    h.AutoRotate=false; h.PlatformStand=true
    flightAttachment.Position=r.CFrame:PointToObjectSpace(r.AssemblyCenterOfMass)
    if followState then
        local goal,velocityOrError,targetMoving,entering=followGoal(followState)
        if not goal then
            stopFollowing(velocityOrError)
            return
        end
        local delta=goal.Position-r.Position
        if entering or delta.Magnitude>45 then
            r.CFrame=goal; r.AssemblyLinearVelocity=Vector3.zero; r.AssemblyAngularVelocity=Vector3.zero
            delta=Vector3.zero
        end
        local correction=delta*8
        local limit=math.max(40,S.FlySpeed*1.5)
        if correction.Magnitude>limit then correction=correction.Unit*limit end
        flight.VectorVelocity=velocityOrError+correction
        local facing=Vector3.new(goal.LookVector.X,0,goal.LookVector.Z)
        if facing.Magnitude>.001 then lastFlightHeading=facing.Unit end
        flightOrientation.CFrame=CFrame.lookAt(Vector3.zero,lastFlightHeading)
        updateFlightAnimation(h,targetMoving)
        return
    end
    local move=Vector3.zero
    if clientFocused and not Input:GetFocusedTextBox() then
        local function down(k) return Input:IsKeyDown(k) and 1 or 0 end
        move=cam.CFrame.LookVector*(down(Enum.KeyCode.W)-down(Enum.KeyCode.S))
            +cam.CFrame.RightVector*(down(Enum.KeyCode.D)-down(Enum.KeyCode.A))
            +Vector3.yAxis*(down(Enum.KeyCode.Space)-down(Enum.KeyCode.LeftControl))
    end
    local moving=move.Magnitude>.01
    flight.VectorVelocity=moving and move.Unit*S.FlySpeed or Vector3.zero
    local heading=Vector3.new(cam.CFrame.LookVector.X,0,cam.CFrame.LookVector.Z)
    if heading.Magnitude>.001 then lastFlightHeading=heading.Unit end
    flightOrientation.CFrame=CFrame.lookAt(Vector3.zero,lastFlightHeading)
    updateFlightAnimation(h,moving)
end

local function stopDesync()
    local d=desync; desync=nil
    if not d then return end
    if flightHumanoid==d.proxyHumanoid then stopFlight() end
    -- Return the original character to the proxy's walked position.
    if d.root.Parent and LP.Character==d.character and d.humanoid.Health>0 then
        local destination=d.proxyRoot.Parent and d.proxyRoot.CFrame or d.origin
        if destination.Position.Y<workspace.FallenPartsDestroyHeight+20 then destination=d.origin end
        d.root.CFrame=destination
        d.root.AssemblyLinearVelocity=Vector3.zero; d.root.AssemblyAngularVelocity=Vector3.zero
    end
    if d.humanoid.Parent then d.humanoid.AutoRotate=d.auto; d.humanoid.PlatformStand=d.platform end
    for part,value in pairs(d.hidden) do if part.Parent then part.LocalTransparencyModifier=value end end
    local camera=workspace.CurrentCamera
    if camera and camera.CameraSubject==d.proxyHumanoid then
        local _,h=rig(LP); camera.CameraSubject=(d.subject and d.subject.Parent and d.subject) or h
        camera.CameraType=d.cameraType
    end
    if d.startMarker then d.startMarker:Destroy() end
    d.proxy:Destroy()
end

local function updateDesync(c,h,r)
    if not S.Desync then stopDesync(); return false end
    if not c or not h or not r then
        stopDesync(); S.Desync=false; refresh(); return false
    end
    if desync and desync.character~=c then stopDesync(); S.Desync=false; refresh(); return false end
    if not desync then startDesync(c,h,r) end
    local d=desync; if not d then return false end
    if not d.proxyRoot.Parent or d.proxyHumanoid.Health<=0 then
        stopDesync(); S.Desync=false; refresh(); return false
    end
    if d.proxyRoot.Position.Y<workspace.FallenPartsDestroyHeight+30 then
        stopDesync(); S.Desync=false; refresh(); notify('Desync ended: local avatar fell out of the map'); return false
    end
    -- Preserve default keyboard, gamepad and touch control direction without replacing Player.Character.
    d.proxyHumanoid.WalkSpeed=h.WalkSpeed
    d.proxyHumanoid.UseJumpPower=h.UseJumpPower; d.proxyHumanoid.JumpPower=h.JumpPower; d.proxyHumanoid.JumpHeight=h.JumpHeight
    if S.Fly then
        d.proxyHumanoid:Move(Vector3.zero,false)
        for _,part in ipairs(d.proxy:GetDescendants()) do
            if part:IsA('BasePart') then
                if d.flightCollision[part]==nil then d.flightCollision[part]=part.CanCollide end
                part.CanCollide=false
            end
        end
        updateFlight(d.proxyHumanoid,d.proxyRoot)
    else
        if flightHumanoid==d.proxyHumanoid then stopFlight() end
        for part,value in pairs(d.flightCollision) do
            if part.Parent then part.CanCollide=value end; d.flightCollision[part]=nil
        end
        d.proxyHumanoid:Move((clientFocused and not Input:GetFocusedTextBox()) and h.MoveDirection or Vector3.zero,false)
    end
    for _,part in ipairs(c:GetDescendants()) do
        if part:IsA('BasePart') then
            if d.hidden[part]==nil then d.hidden[part]=part.LocalTransparencyModifier end
            part.LocalTransparencyModifier=1
        end
    end
    local elapsed=os.clock()-d.start
    local side=math.floor(elapsed)%2==0 and 1 or -1
    local floor=workspace.FallenPartsDestroyHeight
    if floor~=floor or floor==-math.huge then floor=-500 end
    -- Stay above the engine's part-destruction plane while alternating two distant void positions once per second.
    r.CFrame=CFrame.new(d.origin.Position.X+10000*side,floor+100,d.origin.Position.Z+10000)
    r.AssemblyLinearVelocity=Vector3.zero; r.AssemblyAngularVelocity=Vector3.zero
    local camera=workspace.CurrentCamera
    if camera then camera.CameraType=Enum.CameraType.Custom; camera.CameraSubject=d.proxyHumanoid end
    d.startMarker.Transparency=S.DesyncMarker and .2 or 1
    d.startMarker.Color=color(S.DesyncColor); d.startLabel.Enabled=S.DesyncMarker
    d.markerText.TextColor3=color(S.DesyncColor)
    if d.travelTrail.Enabled~=S.DesyncTrail then d.travelTrail:Clear(); d.travelTrail.Enabled=S.DesyncTrail end
    if d.trailColor~=S.DesyncColor then
        d.travelTrail.Color=ColorSequence.new(color(S.DesyncColor)); d.trailColor=S.DesyncColor
    end
    d.highlight.FillColor=color(S.DesyncColor); d.highlight.OutlineColor=color(S.DesyncColor)
    d.aura.Color=color(S.DesyncColor); d.aura.Transparency=.72+math.sin(elapsed*4)*.1
    local diameter=6+math.sin(elapsed*4)*.3; d.aura.Size=Vector3.new(diameter,diameter,diameter)
    return true
end

local function getAimPart(p)
    if not p or p==LP or p.Parent~=Players then return nil end
    local c,_,root=rig(p); if not c then return nil end
    local part=c:FindFirstChild(S.Part) or c:FindFirstChild('Head') or root
    if part and part:IsA('BasePart') then return part end
    return nil
end
local function aimVisible(part,cam)
    local params=RaycastParams.new(); params.FilterType=Enum.RaycastFilterType.Exclude
    local excluded=LP.Character and {LP.Character} or {}
    if desync then table.insert(excluded,desync.proxy) end
    params.FilterDescendantsInstances=excluded
    local hit=workspace:Raycast(cam.CFrame.Position,part.Position-cam.CFrame.Position,params)
    return not hit or hit.Instance:IsDescendantOf(part.Parent)
end
local function eligible(p,cam,center)
    if S.TeamCheck and LP.Team~=nil and p.Team==LP.Team then return end
    local part=getAimPart(p); if not part then return end
    local v,visible=cam:WorldToViewportPoint(part.Position)
    local distance=(Vector2.new(v.X,v.Y)-center).Magnitude
    if not visible or v.Z<=0 or distance>S.FOV then return end
    if S.WallCheck and not aimVisible(part,cam) then return end
    return part,distance
end
local function selectAimTarget(cam,center)
    -- FOV, visibility and team filtering are acquisition checks, not retarget triggers.
    local part=locked and getAimPart(locked)
    if part then return part end
    locked=nil; local best=math.huge
    for _,p in ipairs(Players:GetPlayers()) do
        local hit,d=eligible(p,cam,center)
        if hit and d<best then locked=p; part=hit; best=d end
    end
    return part
end

        BackgroundColor3=pink,TextColor3=Color3.fromRGB(32,13,26),AutoButtonColor=false,Font=Enum.Font.GothamBold,TextSize=13,BorderSizePixel=0,ZIndex=3},card);corner(unlock,8)
    local close=make('TextButton',{Text='x',Position=UDim2.fromOffset(367,20),Size=UDim2.fromOffset(28,28),
        BackgroundTransparency=1,TextColor3=Color3.fromRGB(151,139,165),Font=Enum.Font.Gotham,TextSize=22,ZIndex=4},card)
    local function finish(ok)
        if finished then return end
        finished=true; accepted=ok
        completed:Fire()
    end
    local cancel=function() finish(false) end
    startupEnvironment.VeilsKeyClose=cancel
    connect(input:GetPropertyChangedSignal('Text'),function() status.Text='' end)
    connect(input.Focused,function() tween(inputBorder,{Color=pink}) end)
    connect(input.FocusLost,function(enter) tween(inputBorder,{Color=Color3.fromRGB(48,42,56)}) end)
    local function submit()
        if busy or finished then return end
        local value=input.Text:match('^%s*(.-)%s*$')
        if value~=expectedKey then
            status.Text="That key does not match. Try again."
            tween(inputBorder,{Color=Color3.fromRGB(255,104,137)})
            task.spawn(function()
                for _,x in ipairs({-5,5,-3,3,0}) do
                    if finished then return end
                    tween(input,{Position=UDim2.fromOffset(26+x,150)},.04); task.wait(.04)
                end
            end)
            return
        end
        busy=true; input.TextEditable=false
        unlock.Text='Access granted';tween(unlock,{BackgroundColor3=Color3.fromRGB(134,228,177)})
        status.Text='Welcome to Veils Universe.';status.TextColor3=Color3.fromRGB(134,228,177)
        tween(border,{Color=Color3.fromRGB(134,228,177)})
        task.delay(.45,function() finish(true) end)
    end
    connect(unlock.Activated,submit)
    -- Enter uses the same validation path as the button.
    connect(input.FocusLost,function(enter) if enter then submit() end end)
    connect(unlock.MouseEnter,function() if not busy then tween(unlock,{BackgroundColor3=Color3.fromRGB(255,145,198)}) end end)
    connect(unlock.MouseLeave,function() if not busy then tween(unlock,{BackgroundColor3=pink}) end end)
    connect(close.Activated,cancel)
    connect(root.Destroying,function() finish(false) end)
    tween(card,{Position=UDim2.fromScale(.5,.5)},.3)
    completed.Event:Wait()
    for _,connection in ipairs(connections) do connection:Disconnect() end
    if startupEnvironment.VeilsKeyClose==cancel then startupEnvironment.VeilsKeyClose=nil end
    root:Destroy();completed:Destroy()
    return accepted
end
if not awaitVeilsKey() then return end

local startupOK,startupError=xpcall(bootVeils,function(err)
    return debug.traceback(tostring(err),2)
end)
if startupOK then
    startupEnvironment.VeilsLastError=nil
    if startupOverlay then startupOverlay:Destroy(); startupOverlay=nil end
    print('[Veils Universe] Menu ready')
else
    local message='Veils startup failed at '..startupStage..':\n'..tostring(startupError)
    startupEnvironment.VeilsLastError=message
    warn(message)
    local instance=startupEnvironment.VeilsUniverse
    if type(instance)=='table' and type(instance.Unload)=='function' then pcall(instance.Unload) end
    pcall(showStartupStatus,message,true)
end
